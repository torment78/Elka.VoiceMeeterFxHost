function(capture_between input start_marker end_marker output)
    string(FIND "${input}" "${start_marker}" first)
    if(first LESS 0)
        message(FATAL_ERROR "Plugin failure check source marker missing: ${start_marker}")
    endif()
    string(SUBSTRING "${input}" ${first} -1 tail)
    string(FIND "${tail}" "${end_marker}" last)
    if(last LESS 0)
        message(FATAL_ERROR "Plugin failure check end marker missing: ${end_marker}")
    endif()
    string(SUBSTRING "${tail}" 0 ${last} result)
    set(${output} "${result}" PARENT_SCOPE)
endfunction()
